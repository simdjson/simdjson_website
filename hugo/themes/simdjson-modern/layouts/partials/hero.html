quickstart.cpp 
#include "simdjson.h"
using namespace simdjson;

int main() {
  ondemand::parser parser;
  padded_string json = padded_string::load("twitter.json");
  ondemand::document tweets = parser.iterate(json);
  std::cout << uint64_t(tweets["search_metadata"]["count"])
            << " results." << std::endl;
}
